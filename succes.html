const params = new URLSearchParams(window.location.search); const produitId = params.get("produitId"); const container = document.getElementById("download"); async function verifierAcces() { if (!produitId) { container.innerHTML = "❌ Produit introuvable"; return; } container.innerHTML = "🔄 Vérification du paiement..."; try { const res = await fetch("/.netlify/functions/check-access", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ produitId }) }); const data = await res.json(); if (data.allowed && data.file) { container.innerHTML = ` 📥 Télécharger votre fichier `; } else { container.innerHTML = "❌ Accès refusé ou paiement non validé"; } } catch (error) { console.error(error); container.innerHTML = "❌ Erreur serveur, réessayez"; } } verifierAcces();
